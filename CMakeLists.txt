set(mrubyc_dir "${CMAKE_CURRENT_LIST_DIR}/vendor/mrubyc")

idf_component_register(
    SRC_DIRS "${mrubyc_dir}/src" "${mrubyc_dir}/hal/esp32"
    INCLUDE_DIRS "${mrubyc_dir}/src" "${mrubyc_dir}/hal/esp32"
    REQUIRES freertos
    PRIV_REQUIRES esp_driver_gptimer soc
)

target_sources(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/gptimer_compat.c")

target_compile_definitions(${COMPONENT_LIB} PUBLIC
    MAX_VM_COUNT=${CONFIG_MRUBYC_MAX_VM_COUNT}
    MAX_REGS_SIZE=${CONFIG_MRUBYC_MAX_REGS_SIZE}
    MAX_SYMBOLS_COUNT=${CONFIG_MRUBYC_MAX_SYMBOLS_COUNT}
    MRBC_USE_FLOAT=${CONFIG_MRUBYC_FLOAT_VALUE}
    MRBC_USE_MATH=$<BOOL:${CONFIG_MRUBYC_USE_MATH}>
    MRBC_STRING_SIZE_T=uint${CONFIG_MRUBYC_STRING_SIZE_BITS}_t
    MRBC_USE_STRING_UTF8=$<BOOL:${CONFIG_MRUBYC_USE_STRING_UTF8}>
    MRBC_USE_UNICODE_CASE=$<BOOL:${CONFIG_MRUBYC_USE_UNICODE_CASE}>
    MRBC_SCHEDULER_EXIT=$<BOOL:${CONFIG_MRUBYC_SCHEDULER_EXIT}>
    MRBC_TIMESLICE_TICK_COUNT=${CONFIG_MRUBYC_TIMESLICE_TICK_COUNT}
    MRBC_TICK_UNIT=${CONFIG_MRUBYC_TICK_VALUE}
    MRBC_EXCEPTION_CALL_NEST_LEVEL=${CONFIG_MRUBYC_EXCEPTION_CALL_NEST_LEVEL}
    MRBC_TASK_NAME_LEN=${CONFIG_MRUBYC_TASK_NAME_LEN}
    PRIVATE
    MRBC_KV_SIZE_INIT=${CONFIG_MRUBYC_KV_SIZE_INIT}
    MRBC_KV_SIZE_INCREMENT=${CONFIG_MRUBYC_KV_SIZE_INCREMENT}
)

if(NOT CONFIG_MRUBYC_ALLOC_LIBC)
    target_sources(${COMPONENT_LIB} PRIVATE allocator_config.c)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        MRBC_ALLOC_FLI_BIT_WIDTH=${CONFIG_MRUBYC_ALLOC_FLI_BIT_WIDTH}
        MRBC_ALLOC_SLI_BIT_WIDTH=${CONFIG_MRUBYC_ALLOC_SLI_BIT_WIDTH}
        MRBC_ALLOC_IGNORE_LSBS=${CONFIG_MRUBYC_ALLOC_IGNORE_LSBS}
        MRBC_MIN_MEMORY_BLOCK_SIZE=${CONFIG_MRUBYC_MIN_MEMORY_BLOCK_SIZE}
    )
endif()

target_compile_definitions(${COMPONENT_LIB}
    PUBLIC
    $<$<BOOL:${CONFIG_MRUBYC_ALLOC_PROFILING}>:MRBC_USE_ALLOC_PROF=1>
    $<$<BOOL:${CONFIG_MRUBYC_DISABLE_LEGACY_VALUE_MACROS}>:MRBC_NOT_RECOMMEND_TO_USE=1>
    PRIVATE
    $<$<BOOL:${CONFIG_MRUBYC_SYMBOL_SEARCH_LINEAR}>:MRBC_SYMBOL_SEARCH_LINEAR=1>
    $<$<BOOL:${CONFIG_MRUBYC_NO_STDIO}>:MRBC_NO_STDIO=1>
    $<$<BOOL:${CONFIG_MRUBYC_DEBUG_REGS}>:MRBC_DEBUG_REGS=1>
)

if(CONFIG_MRUBYC_OOM_POOL_STATS OR CONFIG_MRUBYC_PANIC_ON_UNHANDLED_EXCEPTION)
    target_sources(${COMPONENT_LIB} PRIVATE fatal_hooks.c)
    target_compile_options(${COMPONENT_LIB} PRIVATE
        "$<$<COMPILE_LANGUAGE:C>:SHELL:-include \"${CMAKE_CURRENT_LIST_DIR}/fatal_hooks.h\">")
endif()

foreach(option IN ITEMS INT64 NAN_BOXING SUPPORT_OP_EXT ALLOC_LIBC NO_TIMER CONVERT_CRLF)
    if(CONFIG_MRUBYC_${option})
        target_compile_definitions(${COMPONENT_LIB} PUBLIC MRBC_${option}=1)
    endif()
endforeach()

if(NOT CONFIG_MRUBYC_DEBUG)
    target_compile_definitions(${COMPONENT_LIB} PUBLIC NDEBUG)
endif()

target_compile_options(${COMPONENT_LIB} PRIVATE -Wno-error=implicit-fallthrough)
include(CheckCCompilerFlag)
check_c_compiler_flag("-Wunterminated-string-initialization" MRUBYC_HAS_UNTERMINATED_STRING_WARNING)
if(MRUBYC_HAS_UNTERMINATED_STRING_WARNING)
    target_compile_options(${COMPONENT_LIB} PRIVATE -Wno-error=unterminated-string-initialization)
endif()
